+class FakeOrientedBoundingBox3D:
+    """``adsk.core.OrientedBoundingBox3D``: a centre, two directions, three sizes.
+
+    Read from the shipped stub (``adsk/core.py``, ``OrientedBoundingBox3D.create``):
+    ``centerPoint`` is the box's **centre**, the length and width directions carry
+    only a direction ("the magnitude of the vector is ignored"), the width
+    direction "must be perpendicular to the length direction", and the height runs
+    along length x width by the right-hand rule.
+
+    The perpendicularity is enforced rather than waved through: two parallel axes
+    describe no box, and a fake that accepted them could not tell a working
+    ``create_box`` from one that builds a degenerate solid.  A non-finite size or
+    a non-positive one is refused for the same reason - it is a size Fusion would
+    not build either.
+    """
+
+    def __init__(
+        self,
+        center_point: FakePoint3D,
+        length_direction: FakePoint3D,
+        width_direction: FakePoint3D,
+        length: float,
+        width: float,
+        height: float,
+    ):
+        self.centerPoint = center_point
+        self.lengthDirection = FakeVector3D(*_as_unit(length_direction, "length"))
+        self.widthDirection = FakeVector3D(*_as_unit(width_direction, "width"))
+        if abs(_dot(self.lengthDirection, self.widthDirection)) > 1e-9:
+            raise ValueError("widthDirection must be perpendicular to lengthDirection")
+        self.heightDirection = FakeVector3D(
+            *_cross(self.lengthDirection, self.widthDirection)
+        )
+        for name, value in (
+            ("length", length),
+            ("width", width),
+            ("height", height),
+        ):
+            if not math.isfinite(value) or value <= 0:
+                raise ValueError(f"{name} must be a positive finite size: {value!r}")
+        self.length = float(length)
+        self.width = float(width)
+        self.height = float(height)
+
+    @staticmethod
+    def create(
+        centerPoint: FakePoint3D,
+        lengthDirection: FakePoint3D,
+        widthDirection: FakePoint3D,
+        length: float,
+        width: float,
+        height: float,

+            centerPoint, lengthDirection, widthDirection, length, width, height

+    def world_half_extents(self) -> tuple[float, float, float]:
+        """Half the box's size along each **world** axis, after orientation.
+
+        An oriented box's axis-aligned box is the projection of its three edge
+        vectors, which for a box built on the world axes is simply half its size
+        on that axis.
+        """
+        directions = (
+            self.lengthDirection,
+            self.widthDirection,
+            self.heightDirection,
+        )

+            sum(abs(d.coords()[2]) * h for d, h in zip(directions, halves)),

+def _as_unit(vector: FakePoint3D, name: str) -> tuple[float, float, float]:
+    """A direction vector, normalised - Fusion ignores the magnitude."""
+    x, y, z = vector.coords()
+    length = math.sqrt(x * x + y * y + z * z)
+    if length == 0:
+        raise ValueError(f"{name}Direction has no direction: {vector!r}")
+    return (x / length, y / length, z / length)

+def _dot(a: FakePoint3D, b: FakePoint3D) -> float:
+    ax, ay, az = a.coords()
+    bx, by, bz = b.coords()
+    return ax * bx + ay * by + az * bz

+"""``create_box``, driven through the handler: where the box actually lands.
+
+``center_z`` used to mean *base* z.  The orientation matrix was built at
+``center_z + height / 2``, so X and Y were centred on the caller's point while Z
+*started* there: a 10 x 10 x 2 box asked for at ``center_z=1`` came out spanning
+z 1..3, and one asked for with the default ``center_z=0`` sat on the origin plane
+instead of straddling it.  Nothing in the response said where the solid went, so
+nothing noticed - the schema name was the only statement about it, and the schema
+name was wrong.
+
+The fake had no primitives at all, which is why this was measured by hand rather
+than tested: ``TemporaryBRepManager``, ``OrientedBoundingBox3D`` and a base
+feature a body can be added to all had to exist before the handler could run.
+"""
+
+from __future__ import annotations
+
+import pytest
+
+from tests.fakes.design import FakeDesign
+
+
+def _first_body(bound_handler, **kwargs):
+    """Run ``create_box`` and hand back the body that reached the root component.
+
+    ``features=0`` is an empty design: without it the fake gives the root one
+    default body, and ``item(0)`` would be that one rather than the box.
+    """
+    design = FakeDesign(features=0)
+    bound_handler(design).create_box(**kwargs)
+    assert design.rootComponent.bRepBodies.count == 1, "exactly one body was created"
+    return design.rootComponent.bRepBodies.item(0)
+
+
+def test_a_box_is_centred_on_the_point_it_is_given(bound_handler):
+    """The three sizes straddle the centre point, on every axis.
+
+    X and Y already did; Z did not, which is the defect.  All six numbers are
+    asserted together because "centred" is one claim: half the box either side of
+    the point, whichever axis you look along.
+
+    Length and width differ (10 and 6) on purpose: with a square footprint, a
+    handler that swapped the two directions would land on the same box and this
+    test could not tell the two axes apart.
+    """
+    body = _first_body(
+        bound_handler,
+        length=10,
+        width=6,
+        height=2,
+        center_x=1,
+        center_y=2,
+        center_z=1,
+    )
+
+    box = body.boundingBox
+    assert (
+        box.minPoint.x,
+        box.minPoint.y,
+        box.minPoint.z,
+    ) == pytest.approx((-4.0, -1.0, 0.0))
+    assert (
+        box.maxPoint.x,
+        box.maxPoint.y,
+        box.maxPoint.z,
+    ) == pytest.approx((6.0, 5.0, 2.0))
+

+def test_the_default_box_straddles_the_model_origin(bound_handler):
+    """``center_z`` left out means centred on 0, so the box runs -h/2 .. +h/2.
+

+    box used to sit *on* the origin plane and now straddles it, which is what a
+    centre always promised.
+    """
+    body = _first_body(bound_handler, length=10, width=10, height=2)
+
+    assert body.boundingBox.minPoint.z == pytest.approx(-1.0)
+    assert body.boundingBox.maxPoint.z == pytest.approx(1.0)
